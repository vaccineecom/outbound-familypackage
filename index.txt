    (function () {
      const frame = document.getElementById('app-frame');
      const loading = document.getElementById('loading');
      const err = document.getElementById('err');

      // Kiểm tra link đã được cấu hình đúng chưa
      const ok = typeof APPS_SCRIPT_URL === 'string'
        && APPS_SCRIPT_URL.indexOf('https://script.google.com/') === 0
        && APPS_SCRIPT_URL.endsWith('/exec')
        && APPS_SCRIPT_URL.indexOf('XXXXXXXX') === -1;

      if (!ok) {
        loading.style.display = 'none';
        err.style.display = 'block';
        return;
      }

      // Nạp app vào iframe
      frame.src = APPS_SCRIPT_URL;

      // Ẩn màn hình chờ khi iframe tải xong
      frame.addEventListener('load', function () {
        loading.classList.add('hide');
        setTimeout(function () { loading.style.display = 'none'; }, 450);
      });

      // Phòng trường hợp sự kiện load không kích hoạt: ẩn spinner sau 8 giây

        loading.classList.add('hide');
        setTimeout(function () { loading.style.display = 'none'; }, 450);
      }, 8000);
    })();
